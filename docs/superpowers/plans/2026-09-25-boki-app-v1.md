# 簿記2級 学習アプリ v1 実装計画

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Excel+Anki の簿記2級学習管理を、GitHub Pages で動く PWA（間隔反復・仕訳演習・日次記録・ダッシュボード・JSON同期）として構築する。

**Architecture:** ビルド無しの素の HTML/JS（ES modules）。データは IndexedDB。`srs.js` `scoring.js` は純関数で重点テスト。画面は `js/views/*.js` に1画面1ファイル、`app.js` がハッシュルーティングで切替。テストは `test/index.html` をブラウザで開く自作ランナー（Node 不要）。

**Tech Stack:** HTML/CSS/JavaScript(ES2020+ modules), IndexedDB, Service Worker, Python 3 + openpyxl（Excel→seed 変換のみ）, GitHub Pages。

**Spec:** `docs/superpowers/specs/2026-09-25-boki-app-design.md`

## Global Constraints

- 外部ライブラリ・CDN 依存無し。npm/Node 不使用（会社PCに無い）。
- 会社PCへ新規ソフト導入時はユーザー確認必須。
- IndexedDB のみ。localStorage 縮退無し。
- 評価ボタン文言: `できなかった／迷ったが正解／できた／即答できた`（rating 1〜4）。
- 復習ボタン下に常時表示: `一部しか答えられなかったら「できなかった」`
- 正答率 = 正解÷(正解+不正解)。分母0は「データ不足」。未回答・要確認は分母除外。
- ID 形式: カード `FC-W01-0001`、日次 `DL-YYYYMMDD-区分NN`、問題 `Q-0001`、回答 `AL-<timestamp>`、ロードマップ `RM-01`。
- 間違い原因: `知識不足／科目判断ミス／読み違い／計算ミス／桁ミス／時間切れ／回答範囲不足`
- interval 上限 180 日、ease 下限 1.3、初期 ease 2.5。
- スマホ幅 360px 想定、ボタン高さ 48px 以上。
- 個人情報・勤務先情報を保存しない。問題文は架空企業名・架空金額。
- ローカル確認: `boki-app/` で `python -m http.server 8000` → `http://localhost:8000/`。
- テスト実行: ブラウザで `http://localhost:8000/test/` を開き fail 0 を確認。
- 各タスク末尾でコミット。Conventional Commits。
- OneDrive 上の xlsx は直読みで PermissionError になるため、`$TEMP` にコピーして読む。

```js
const tests = [];
export function test(name, fn) { tests.push({ name, fn }); }
function fail(msg) { throw new Error(msg); }
export const assert = {
  ok(v, msg = 'expected truthy') { if (!v) fail(msg); },
  equal(a, b, msg = '') { if (a !== b) fail(`${msg} expected ${JSON.stringify(b)} got ${JSON.stringify(a)}`); },
  deepEqual(a, b, msg = '') {
    const ja = JSON.stringify(a), jb = JSON.stringify(b);
    if (ja !== jb) fail(`${msg} expected ${jb} got ${ja}`);
  },
  async throws(fn, msg = 'expected throw') {
    try { await fn(); } catch { return; }
    fail(msg);
  },
};
export async function run() {
  const out = document.getElementById('out');
  let pass = 0, failCount = 0;
  for (const t of tests) {
    const li = document.createElement('li');
    try { await t.fn(); li.className = 'pass'; li.textContent = `✔ ${t.name}`; pass++; }
    catch (e) { li.className = 'fail'; li.textContent = `✘ ${t.name}: ${e.message}`; failCount++; }
    out.appendChild(li);
  }
  document.getElementById('summary').textContent = `pass ${pass} / fail ${failCount}`;
  document.title = failCount ? `FAIL ${failCount}` : 'ALL PASS';
}
```

```js
export const RATING = { AGAIN: 1, HARD: 2, GOOD: 3, EASY: 4 };
export const RATING_LABEL = { 1: 'できなかった', 2: '迷ったが正解', 3: 'できた', 4: '即答できた' };
export const RATING_HINT = '一部しか答えられなかったら「できなかった」';
const MAX_INTERVAL = 180, MIN_EASE = 1.3;
const round2 = x => Math.round(x * 100) / 100;

export function addDays(iso, n) {
  const d = new Date(iso + 'T00:00:00Z'); d.setUTCDate(d.getUTCDate() + n);
  return d.toISOString().slice(0, 10);
}
export function newSrsState() {
  return { due: null, interval: 0, ease: 2.5, reps: 0, streak: 0, lastResult: null, lastReviewed: null };
}
export function applyRating(state, rating, today) {
  let { interval, ease, streak } = state;
  if (rating === RATING.AGAIN) { interval = 0; ease = Math.max(MIN_EASE, round2(ease - 0.2)); streak = 0; }
  else if (rating === RATING.HARD) { interval = Math.max(1, Math.round(interval * 1.2)); ease = Math.max(MIN_EASE, round2(ease - 0.15)); streak++; }
  else if (rating === RATING.GOOD) { interval = interval === 0 ? 1 : interval === 1 ? 3 : Math.round(interval * ease); streak++; }
  else if (rating === RATING.EASY) { interval = interval === 0 ? 4 : Math.max(interval + 1, Math.round(interval * ease * 1.3)); ease = round2(ease + 0.15); streak++; }
  else throw new Error('invalid rating ' + rating);
  interval = Math.min(MAX_INTERVAL, interval);
  return { ...state, interval, ease, streak, reps: state.reps + 1, due: addDays(today, interval), lastResult: rating, lastReviewed: today };
}
export function cardStatus(s) { return s.reps === 0 ? '新規' : s.interval < 21 ? '学習中' : '定着'; }
export function isDue(s, today) { return !s.due || s.due <= today; }
const PRI = { A: 0, B: 1, C: 2 };
export function dueCards(cards, today) {
  return cards.filter(c => c.status !== '停止' && isDue(c, today))
    .sort((a, b) => (PRI[a.priority] ?? 9) - (PRI[b.priority] ?? 9) || String(a.due ?? '').localeCompare(String(b.due ?? '')));
}
```

```js
export const CAUSES = ['知識不足', '科目判断ミス', '読み違い', '計算ミス', '桁ミス', '時間切れ', '回答範囲不足'];

export function normalizeNumber(v) {
  if (v === null || v === undefined) return null;
  if (typeof v === 'number') return Number.isFinite(v) ? v : null;
  const s = String(v).replace(/[０-９]/g, c => String.fromCharCode(c.charCodeAt(0) - 0xFEE0)).replace(/[,\s　円]/g, '');
  if (s === '' || !/^-?\d+(\.\d+)?$/.test(s)) return null;
  return Number(s);
}
const clean = s => String(s ?? '').replace(/[\s　]/g, '');
function lines(side) {
  return (side ?? []).map(r => ({ acct: clean(r.acct), amt: normalizeNumber(r.amt) })).filter(r => r.acct || r.amt !== null);
}
const key = rs => rs.map(r => `${r.acct}:${r.amt}`).sort().join('|');
const accts = rs => rs.map(r => r.acct).sort().join('|');
const sum = rs => rs.reduce((a, r) => a + (r.amt ?? 0), 0);

export function gradeJournal(answer, my) {
  const ad = lines(answer.debit), ac = lines(answer.credit), md = lines(my.debit), mc = lines(my.credit);
  if (md.length === 0 && mc.length === 0) return { correct: false, kind: '未入力' };
  if (key(ad) === key(md) && key(ac) === key(mc)) return { correct: true, kind: '正解' };
  if (sum(md) !== sum(mc)) return { correct: false, kind: '貸借不一致' };
  if (accts(ad) === accts(md) && accts(ac) === accts(mc)) return { correct: false, kind: '金額ミス' };
  return { correct: false, kind: '科目ミス' };
}
export function gradeExact(answer, my) { return clean(answer) === clean(my); }
```

```js
export const STORES = ['settings', 'roadmap', 'cards', 'reviewLogs', 'questions', 'answerLogs', 'dailyLogs', 'accounts'];
const KEYS = { accounts: 'name' };
const keyOf = s => KEYS[s] ?? 'id';
const VERSION = 1;

const req = r => new Promise((res, rej) => { r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
const done = tx => new Promise((res, rej) => { tx.oncomplete = res; tx.onerror = () => rej(tx.error); tx.onabort = () => rej(tx.error); });

export function openDb(name = 'boki') {
  if (!('indexedDB' in globalThis)) return Promise.reject(new Error('IndexedDB unavailable'));
  const r = indexedDB.open(name, VERSION);
  r.onupgradeneeded = () => { const db = r.result; for (const s of STORES) if (!db.objectStoreNames.contains(s)) db.createObjectStore(s, { keyPath: keyOf(s) }); };
  return req(r);
}
export function deleteDb(name) { return req(indexedDB.deleteDatabase(name)); }

export async function put(db, store, obj, opts = {}) {
  const o = opts.keepUpdatedAt && obj.updatedAt ? obj : { ...obj, updatedAt: new Date().toISOString() };
  const tx = db.transaction(store, 'readwrite'); tx.objectStore(store).put(o); await done(tx); return o;
}
export async function putAll(db, store, arr, opts = {}) {
  const now = new Date().toISOString(); const tx = db.transaction(store, 'readwrite'); const os = tx.objectStore(store);
  for (const obj of arr) os.put(opts.keepUpdatedAt && obj.updatedAt ? obj : { ...obj, updatedAt: now });
  await done(tx);
}
export function get(db, store, key) { return req(db.transaction(store).objectStore(store).get(key)); }
export function getAll(db, store) { return req(db.transaction(store).objectStore(store).getAll()); }
export function count(db, store) { return req(db.transaction(store).objectStore(store).count()); }
export async function del(db, store, key) { const tx = db.transaction(store, 'readwrite'); tx.objectStore(store).delete(key); await done(tx); }
export async function clear(db, store) { const tx = db.transaction(store, 'readwrite'); tx.objectStore(store).clear(); await done(tx); }

export async function exportAll(db) {
  const stores = {}; for (const s of STORES) stores[s] = await getAll(db, s);
  return { version: VERSION, exportedAt: new Date().toISOString(), stores };
}
export function validateBackup(d) {
  if (!d || typeof d !== 'object') return { ok: false, error: 'JSONではありません' };
  if (d.version !== VERSION) return { ok: false, error: `version不一致（期待${VERSION}、実際${d.version}）` };
  if (!d.stores || typeof d.stores !== 'object') return { ok: false, error: 'storesがありません' };
  for (const s of STORES) if (!Array.isArray(d.stores[s])) return { ok: false, error: `ストア「${s}」がありません` };
  return { ok: true };
}
export async function importAll(db, data, mode) {
  const v = validateBackup(data); if (!v.ok) throw new Error(v.error);
  for (const s of STORES) {
    if (mode === 'overwrite') { await clear(db, s); await putAll(db, s, data.stores[s], { keepUpdatedAt: true }); continue; }
    const k = keyOf(s); const local = new Map((await getAll(db, s)).map(o => [o[k], o]));
    const merged = data.stores[s].filter(r => { const l = local.get(r[k]); return !l || String(r.updatedAt ?? '') > String(l.updatedAt ?? ''); });
    await putAll(db, s, merged, { keepUpdatedAt: true });
  }
}
```

```js
      el('p', { class: 'muted small' }, RATING_HINT));
    root.replaceChildren(el('div', { class: 'card' },
      el('p', { class: 'muted' }, `${firstRating.size + 1} / ${total}　${c.field}・${c.topic}・${c.priority}　${cardStatus(c)}`),
      el('h3', {}, c.front),
      el('button', { class: 'primary block', onclick: e => { back.hidden = false; e.target.hidden = true; } }, '答えを見る'), back));
  };

    const s = applyRating(c, r, today);
    const updated = { ...c, ...s, status: c.status === '停止' ? c.status : cardStatus(s) };
    await put(ctx.db, 'cards', updated);
    await put(ctx.db, 'reviewLogs', { id: `RL-${Date.now()}-${c.id}`, cardId: c.id, at: new Date().toISOString(), rating: r, prevInterval: c.interval, newInterval: s.interval, elapsedSec: Math.round((Date.now() - cardStart) / 1000) });
    pending.shift();

    if (r === 1) pending.push(updated);
    next();

    const results = [...firstRating].map(([cardId, fr]) => ({ cardId, finalRating: fr }));
    const sum = summarizeReview(results, (Date.now() - startAt) / 1000);
    const row = await saveDailyLog(ctx.db, ctx.settings, { date: today, code: 'CARD', sessionType: '単語帳', field: queue[0]?.field, topic: 'カード復習', priority: 'A', content: `カード復習 ${sum.q}枚`, ...sum, cause: sum.wrong ? '知識不足' : '' });

  };
  next();
}
```

```python
import zlib, struct
def png(size, rgb=(21, 101, 192)):
    raw = b''.join(b'\x00' + bytes(rgb) * size for _ in range(size))
    def chunk(t, d): return struct.pack('>I', len(d)) + t + d + struct.pack('>I', zlib.crc32(t + d) & 0xffffffff)
    return b'\x89PNG\r\n\x1a\n' + chunk(b'IHDR', struct.pack('>IIBBBBB', size, size, 8, 2, 0, 0, 0)) + chunk(b'IDAT', zlib.compress(raw)) + chunk(b'IEND', b'')
for s in (192, 512): open(f'icon-{s}.png', 'wb').write(png(s))
```

```js
self.addEventListener('install', e => e.waitUntil(caches.open(V).then(c => c.addAll(ASSETS)).then(() => self.skipWaiting())));
self.addEventListener('activate', e => e.waitUntil(caches.keys().then(ks => Promise.all(ks.filter(k => k !== V).map(k => caches.delete(k)))).then(() => self.clients.claim())));
self.addEventListener('fetch', e => {
  if (e.request.method !== 'GET' || !e.request.url.startsWith(self.location.origin)) return;
  e.respondWith(fetch(e.request).then(r => { const copy = r.clone(); caches.open(V).then(c => c.put(e.request, copy)); return r; }).catch(() => caches.match(e.request)));
});
```
